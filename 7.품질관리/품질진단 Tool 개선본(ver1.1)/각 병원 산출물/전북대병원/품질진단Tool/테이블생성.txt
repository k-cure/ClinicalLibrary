★★ YourSchemaName => Ctrl+H 를 눌러서 자신의 스키마명(유저명)으로 변경 후 실행시켜주세요.

CREATE TABLE YourSchemaName.RESULT_QUALITY_CHECK (
    RULE_ID VARCHAR2(30),
    TB_NM VARCHAR2(50),
    COL_NM VARCHAR2(50),
    TOT_CNT VARCHAR2(50),
    ERR_CNT VARCHAR2(50),
    ERR_VALUE VARCHAR2(4000)
);

CREATE SEQUENCE YourSchemaName.QUALITY_CHECK_SEQ
START WITH 1
INCREMENT BY 1
NOCACHE
NOCYCLE;

CREATE TABLE YourSchemaName.QUALITY_CHECK_ERROR_LOG (
    SEQ NUMBER(19, 0) PRIMARY KEY,
    CHK_ID VARCHAR2(30),
    CHK_TIME TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ERROR_MESSAGE VARCHAR2(4000)
);
