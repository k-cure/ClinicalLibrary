BEGIN
    -- 테이블 생성
    IF OBJECT_ID('dbo.result_quality_check', 'U') IS NULL
    BEGIN
        CREATE TABLE dbo.result_quality_check (
            RULE_ID VARCHAR(30),
            TB_NM VARCHAR(50),
            COL_NM VARCHAR(50),
            TOT_CNT VARCHAR(50),
            ERR_CNT VARCHAR(50),
            ERR_VALUE TEXT
        );
    END;
    IF OBJECT_ID('dbo.quality_check_error_log', 'U') IS NULL
    BEGIN
        CREATE TABLE dbo.quality_check_error_log (
            SEQ INT IDENTITY(1,1) PRIMARY KEY,
            CHK_ID VARCHAR(30),
            CHK_TIME DATETIME DEFAULT GETDATE(),
            ERROR_MESSAGE TEXT
        );
    END;
END;
BEGIN
    DELETE FROM RESULT_QUALITY_CHECK WHERE 1=1;
END;
BEGIN TRY
    WITH query_result AS (
        SELECT 'ID1002' AS RULE_ID,  
   'PERSON' AS TB_NM,    
   'PERSON_ID' AS COL_NM, 
   COUNT(*) AS TOT_CNT   
   FROM YourSchemaName.PERSON
 
    )
    SELECT
        @RULE_ID_VAL = RULE_ID,
        @TB_NM_VAL = TB_NM,
        @COL_NM_VAL = COL_NM,
        @TOT_CNT_VAL = TOT_CNT
    FROM query_result;
    -- 결과를 테이블에 삽입
    INSERT INTO dbo.result_quality_check (RULE_ID, TB_NM, COL_NM, TOT_CNT)
    VALUES (@RULE_ID_VAL, @TB_NM_VAL, @COL_NM_VAL, @TOT_CNT_VAL);
END TRY
BEGIN CATCH
    -- 오류 발생 시 에러 로그에 기록
    INSERT INTO dbo.quality_check_error_log (CHK_ID, ERROR_MESSAGE)
    VALUES (@RULE_ID_VAL, ERROR_MESSAGE());
END CATCH;
 COMMIT;
