) XX

        -- 오류 발생 시 에러 로그에 기록
        DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
        BEGIN
            INSERT INTO quality_check_error_log (CHK_ID, ERROR_MESSAGE)
            VALUES (RULE_ID_VAL, ERROR_MESSAGE());
        END;

        -- 쿼리 결과를 변수에 저장
       SELECT ERR_VALUE, RULE_ID, TB_NM, COL_NM FROM (
 SELECT DISTINCT SUBSTRING(CAST(DISEASE_STATUS_CONCEPT_ID AS VARCHAR(50)),1,30) AS ERR_VALUE, 'ID3338' AS RULE_ID,  
   'SPECIMEN' AS TB_NM,    
   'DISEASE_STATUS_CONCEPT_ID' AS COL_NM 
FROM YourSchemaName.SPECIMEN A
WHERE CASE WHEN NOT EXISTS 
(SELECT 1 FROM YourSchemaName.CONCEPT B WHERE B.CONCEPT_ID = A.DISEASE_STATUS_CONCEPT_ID)  
THEN 'Y' ELSE 'N' END = 'Y' 
AND ( DISEASE_STATUS_CONCEPT_ID IS NOT NULL AND LENGTH(DISEASE_STATUS_CONCEPT_ID)>0 ) 
AND DISEASE_STATUS_CONCEPT_ID != 0
LIMIT 3
) XX

        INTO ERR_VALUE_VAL, RULE_ID_VAL, TB_NM_VAL, COL_NM_VAL;

        -- 결과를 테이블에 삽입
        update result_quality_check set ERR_VALUE = ERR_VALUE_VAL where RULE_ID = RULE_ID_VAL;
    END;
    COMMIT;
    END;
