BEGIN

    IF NOT EXISTS (SELECT * FROM SYS.TABLES WHERE UPPER(TABLE_NAME) = 'RESULT_QUALITY_CHECK') THEN
        EXECUTE IMMEDIATE 'CREATE TABLE result_quality_check (
            RULE_ID VARCHAR(30),
            TB_NM VARCHAR(50),
            COL_NM VARCHAR(50),
            TOT_CNT VARCHAR(50),
            ERR_CNT VARCHAR(50),
            ERR_VALUE TEXT
        )';
    END IF;
    COMMIT;
    IF NOT EXISTS (SELECT * FROM SYS.SEQUENCES WHERE UPPER(SEQUENCE_NAME) = 'QUALITY_CHECK_SEQ') THEN
        EXECUTE IMMEDIATE 'CREATE SEQUENCE quality_check_seq
            START WITH 1
            INCREMENT BY 1
            CACHE 10
            CYCLE
        )';
    END IF;
    COMMIT;
    IF NOT EXISTS (SELECT * FROM SYS.TABLES WHERE UPPER(TABLE_NAME) = 'QUALITY_CHECK_ERROR_LOG') THEN
        EXECUTE IMMEDIATE 'CREATE TABLE quality_check_error_log (
            SEQ BIGINT PRIMARY KEY,
            CHK_ID VARCHAR(30),
            CHK_TIME TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            ERROR_MESSAGE TEXT
        )';
    END IF;
    COMMIT;
END;

        RULE_ID_VAL, TB_NM_VAL, COL_NM_VAL, TOT_CNT_VAL

    -- 결과를 테이블에 삽입
