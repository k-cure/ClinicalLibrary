CREATE PROCEDURE quality_check_err_cnt

    -- 변수 선언

    DECLARE @TB_NM_VAL VARCHAR(50);
    DECLARE @COL_NM_VAL VARCHAR(50);
    DECLARE @TOT_CNT_VAL VARCHAR(50);
    DECLARE @ERR_CNT_VAL VARCHAR(50);
    DECLARE @ERR_VALUE_VAL VARCHAR(4000);
BEGIN
    -- 테이블 생성
    IF OBJECT_ID('dbo.result_quality_check', 'U') IS NULL
    BEGIN
        CREATE TABLE dbo.result_quality_check (
            RULE_ID VARCHAR(30),
            TB_NM VARCHAR(50),
            COL_NM VARCHAR(50),
            TOT_CNT VARCHAR(50),
            ERR_CNT VARCHAR(50),
            ERR_VALUE TEXT
        );
    END;
    IF OBJECT_ID('dbo.quality_check_error_log', 'U') IS NULL
    BEGIN
        CREATE TABLE dbo.quality_check_error_log (
            SEQ INT IDENTITY(1,1) PRIMARY KEY,
            CHK_ID VARCHAR(30),
            CHK_TIME DATETIME DEFAULT GETDATE(),
            ERROR_MESSAGE TEXT
        );
    END;
END;
BEGIN
    UPDATE RESULT_QUALITY_CHECK SET ERR_CNT = NULL, ERR_VALUE = NULL WHERE 1=1;
END;
BEGIN TRY
    WITH query_result AS (
        SELECT 'ID3001' AS RULE_ID,  
   'PERSON' AS TB_NM,    
   'PERSON_ID' AS COL_NM, 
	COUNT(*) AS ERR_CNT   
    FROM YourSchemaName.PERSON A
    WHERE PERSON_ID IS NULL 
    )
    SELECT
        @RULE_ID_VAL = RULE_ID,
        @TB_NM_VAL = TB_NM,
        @COL_NM_VAL = COL_NM,
        @ERR_CNT_VAL = ERR_CNT
    FROM query_result;
    -- 결과를 테이블에 삽입
    UPDATE dbo.result_quality_check SET ERR_CNT = @ERR_CNT_VAL WHERE RULE_ID = @RULE_ID_VAL;
END TRY
BEGIN CATCH
    -- 오류 발생 시 에러 로그에 기록
    INSERT INTO dbo.quality_check_error_log (CHK_ID, ERROR_MESSAGE)
    VALUES (@RULE_ID_VAL, ERROR_MESSAGE());
END CATCH;
COMMIT;
